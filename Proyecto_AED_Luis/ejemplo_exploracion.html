resumen_cuant_loc <- function(df, digits = 2, trim_pct = 0.05) {
  df %>%
    select(where(is.numeric)) %>%
    pivot_longer(cols = everything(), names_to = "Variable", values_to = "Valor") %>%
    group_by(Variable) %>%
    summarise(
      N_Obs       = n(),
      N_NA        = sum(is.na(Valor)),

      # Tendencia Central / Localización
      Media       = round(mean(Valor, na.rm = TRUE), digits),
      Media_Trim  = round(mean(Valor, trim = trim_pct, na.rm = TRUE), digits),
      Mediana     = round(median(Valor, na.rm = TRUE), digits),
      
      # Posición / Cuantiles
      Min         = round(min(Valor, na.rm = TRUE), digits),
      P10         = round(quantile(Valor, 0.10, na.rm = TRUE), digits),
      P90         = round(quantile(Valor, 0.90, na.rm = TRUE), digits),
      Max         = round(max(Valor, na.rm = TRUE), digits),
      
      .groups     = "drop"
    )
}

resumen_cuant_disp <- function(df, digits = 3) {
  df %>%
    select(where(is.numeric)) %>%
    pivot_longer(cols = everything(), names_to = "Variable", values_to = "Valor") %>%
    group_by(Variable) %>%
    summarise(

      # Dispersión Paramétrica
      Varianza    = round(var(Valor, na.rm = TRUE), digits),
      SD          = round(sd(Valor, na.rm = TRUE), digits),
      Rango       = round(max(Valor, na.rm = TRUE) - min(Valor, na.rm = TRUE), digits),
      
      # Dispersión Robusta (Inmune a outliers)
      IQR         = round(IQR(Valor, na.rm = TRUE), digits),
      MAD         = round(mad(Valor, na.rm = TRUE), digits),
      
      # Dispersión Relativa (%)
      CV_Pct      = round((sd(Valor, na.rm = TRUE) / mean(Valor, na.rm = TRUE)) * 100, digits),
      
      .groups     = "drop"
    )
}

Parece que hay valores atípicos; utilizaremos la regla intercuartílica para detectarlos.

obtener_outliers_iqr <- function(df, variable, k = 1.5) {
  v <- df[[variable]]
  
  q1 <- quantile(v, 0.25, na.rm = TRUE)
  q3 <- quantile(v, 0.75, na.rm = TRUE)
  iqr_val <- IQR(v, na.rm = TRUE)
  
  # No hay límites útiles si la variable no presenta variación.
  if (iqr_val == 0) {
    return(df[0, , drop = FALSE])
  }
  
  limite_inferior <- q1 - k * iqr_val
  limite_superior <- q3 + k * iqr_val
  
  df <- df %>%
    filter(!is.na(.data[[variable]])) %>%
    filter(.data[[variable]] < limite_inferior | .data[[variable]] > limite_superior) %>%
    select(Provincia, all_of(variable)) %>%
    mutate(
      IQR_Ref      = round(iqr_val, 2),
      Diferencia   = round(abs(.data[[variable]] - q1), 2),
      Tipo_Outlier = if_else(.data[[variable]] < limite_inferior, "Inferior", "Superior")
    )
  return(df)
}
